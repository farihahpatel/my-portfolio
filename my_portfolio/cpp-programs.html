1. Grade CalculatorReads marks (0-100) and prints a grade using an if-else ladder. 
#include <iostream>
using namespace std;

int main()
{
    int marks;
    cout << "Enter student's marks (0-100): ";
    cin >> marks;

    if (marks < 0 || marks > 100)
        cout << "Invalid marks! Marks should be between 0 and 100.";
    else if (marks >= 90)
        cout << "Grade A";
    else if (marks >= 80)
        cout << "Grade B";
    else if (marks >= 70)
        cout << "Grade C";
    else if (marks >= 60)
        cout << "Grade D";
    else if (marks >= 50)
        cout << "Grade E";
    else
        cout << "Fail";
}
2. Electricity Bill CalculatorCalculates a bill using slab rates for units consumed, then adds 18% GST. 
#include <iostream>
using namespace std;

int main()
{
    int units;
    float bill, gst, totalBill;

    cout << "Enter electricity units consumed: ";
    cin >> units;

    if (units <= 100)
    {
        bill = units * 5;
    }
    else if (units <= 200)
    {
        bill = (100 * 5) + ((units - 100) * 7);
    }
    else
    {
        bill = (100 * 5) + (100 * 7) + ((units - 200) * 10);
    }

    gst = bill * 0.18;
    totalBill = bill + gst;
    cout << "\nTotal Bill = Rs. " << totalBill;
}
3. Maximum, Minimum and AverageUses three separate functions to find the largest, smallest and average of 4 numbers. 
#include <iostream>
using namespace std;

// Function to find maximum
int maximum(int a, int b, int c, int d)
{
    int max = a;

    if (b > max)
        max = b;
    if (c > max)
        max = c;
    if (d > max)
        max = d;

    return max;
}

// Function to find minimum
int minimum(int a, int b, int c, int d)
{
    int min = a;

    if (b < min)
        min = b;
    if (c < min)
        min = c;
    if (d < min)
        min = d;

    return min;
}

// Function to find average
float average(int a, int b, int c, int d)
{
    return (a + b + c + d) / 4.0;
}

int main()
{
    int n1, n2, n3, n4;

    cout << "Enter 4 numbers: ";
    cin >> n1 >> n2 >> n3 >> n4;

    cout << "Maximum = " << maximum(n1, n2, n3, n4) << endl;
    cout << "Minimum = " << minimum(n1, n2, n3, n4) << endl;
    cout << "Average = " << average(n1, n2, n3, n4) << endl;

    return 0;
}
4. Leap Year CheckerChecks whether a year is a leap year using the divisibility rules for 4, 100 and 400. 
#include <iostream>
using namespace std;

int main()
{
    int year;

    cout << "Enter a year: ";
    cin >> year;

    if ((year % 400 == 0) || (year % 4 == 0 && year % 100 != 0))
    {
        cout << year << " is a Leap Year.";
    }
    else
    {
        cout << year << " is not a Leap Year.";
    }

    return 0;
}

6. Menu-Driven Program using switchA menu with three options: even or odd, positive or negative, and leap year. 
#include <iostream>
using namespace std;

int main()
{
    int choice, num, year;
    cout << "1. Even or Odd" << endl;
    cout << "2. Positive or Negative" << endl;
    cout << "3. Leap Year or Not" << endl;
    cout << "Enter your choice: ";
    cin >> choice;
    switch (choice)
    {
        case 1:
            cout << "Enter a number: ";
            cin >> num;

            if (num % 2 == 0)
                cout << num << " is Even.";
            else
                cout << num << " is Odd.";
            break;
        case 2:
            cout << "Enter a number: ";
            cin >> num;
            if (num >= 0)
                cout << num << " is Positive.";
            else if (num < 0)
                cout << num << " is Negative.";
            break;
        case 3:
            cout << "Enter a year: ";
            cin >> year;

            if ((year % 400 == 0) || (year % 4 == 0 && year % 100 != 0))
                cout << year << " is a Leap Year.";
            else
                cout << year << " is not a Leap Year.";
            break;
        default:
            cout << "Invalid Choice!";
    }
}
7. Student Class (Marks and Percentage)A class that accepts a student's name, roll number and 5 marks, then displays them with the percentage. 
#include <iostream>
using namespace std;

class Student
{
    string name;
    int rollNo;
    float marks1, marks2, marks3, marks4, marks5;
public:
    void accept()
    {
        cout << "Enter student name: " << endl;
        cin >> name;
        cout << "Enter roll number: " << endl;
        cin >> rollNo;
        cout << "Enter marks of 5 subjects: " << endl;
        cin >> marks1 >> marks2 >> marks3 >> marks4 >> marks5;
    }
    void display()
    {
        cout << "Student Details" << endl;
        cout << "Name = " << name << endl;
        cout << "Roll No = " << rollNo << endl;
        cout << "Marks = " << marks1 << " " << marks2 << " "
             << marks3 << " " << marks4 << " " << marks5 << endl;
    }

    void percentage()
    {
        float total = marks1 + marks2 + marks3 + marks4 + marks5;
        float percentage = total / 5;

        cout << "Percentage = " << percentage << "%" << endl;
    }
};

int main()
{
    Student s;
    s.accept();
    s.display();
    s.percentage();
}
8. Bank Account ClassA class with functions to accept account details, deposit, withdraw (with a balance check) and display the balance. 
#include <iostream>
using namespace std;

class BankAccount
{
    int accountNumber;
    string accountHolder;
    double balance;
public:
    void accept()
    {
        cout << "Enter account number: ";
        cin >> accountNumber;
        cout << "Enter account holder name: ";
        cin >> accountHolder;
        cout << "Enter initial balance: ";
        cin >> balance;
    }
    void deposit()
    {
        double amount;
        cout << "Enter amount to deposit: ";
        cin >> amount;
        balance = balance + amount;
    }

    void withdraw()
    {
        double amount;
        cout << "Enter amount to withdraw: ";
        cin >> amount;
        if (amount <= balance)
        {
            balance = balance - amount;
            cout << "Amount withdrawn successfully." << endl;
        }
        else
        {
            cout << "Insufficient balance!" << endl;
        }
    }
    void displayBalance()
    {
        cout << "Balance = Rs. " << balance << endl;
    }
};

int main()
{
    BankAccount b;
    b.accept();
    b.deposit();
    b.withdraw();
    b.displayBalance();
}
9. Student Class with Constructor and GradeUses a constructor to set student details, then calculates the total, percentage and grade. 
#include <iostream>
using namespace std;

class Student
{
    string name;
    int rollNo;
    int marks1, marks2, marks3, marks4, marks5;
public:
    Student(string n, int r, int m1, int m2, int m3, int m4, int m5)
    {
        name = n;
        rollNo = r;
        marks1 = m1;
        marks2 = m2;
        marks3 = m3;
        marks4 = m4;
        marks5 = m5;
    }
    int total()
    {
        return marks1 + marks2 + marks3 + marks4 + marks5;
    }
    float percentage()
    {
        return total() / 5.0;
    }
    void grade()
    {
        float p = percentage();
        if (p >= 90)
            cout << "Grade A" << endl;
        else if (p >= 80)
            cout << "Grade B" << endl;
        else if (p >= 70)
            cout << "Grade C" << endl;
        else if (p >= 60)
            cout << "Grade D" << endl;
        else if (p >= 50)
            cout << "Grade E" << endl;
        else
            cout << "Fail" << endl;
    }

    void display()
    {
        cout << "Name = " << name << endl;
        cout << "Roll No = " << rollNo << endl;
        cout << "Total = " << total() << endl;
        cout << "Percentage = " << percentage() << "%" << endl;
        grade();
    }
};

int main()
{
    Student s("Farihah", 86, 80, 89, 91, 65, 90);
    s.display();
}
10. Library Book SearchStores book IDs in an array and uses a loop to search for a given ID. 
#include <iostream>
using namespace std;

int main()
{
    int n, searchID, i;
    bool found = false;

    cout << "Enter number of books: ";
    cin >> n;

    int bookID[n];

    cout << "Enter Book IDs:\n";
    for (i = 0; i < n; i++)
    {
        cin >> bookID[i];
    }

    cout << "Enter Book ID to search: ";
    cin >> searchID;

    for (i = 0; i < n; i++)
    {
        if (bookID[i] == searchID)
        {
            found = true;
            break;
        }
    }

    if (found)
        cout << "Book ID is available.";
    else
        cout << "Book ID is not available.";

    return 0;
}
